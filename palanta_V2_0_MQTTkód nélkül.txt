// --- EEPROM betöltés/mentés ---
void setDefaultsIP() {
  cfg.ip[0]=192; cfg.ip[1]=168; cfg.ip[2]=1; cfg.ip[3]=50;
  cfg.gw[0]=192; cfg.gw[1]=168; cfg.gw[2]=1; cfg.gw[3]=1;
  cfg.sn[0]=255; cfg.sn[1]=255; cfg.sn[2]=255; cfg.sn[3]=0;
}
void loadSettings() {
  EEPROM.begin(EEPROM_SIZE);
  EEPROM.get(0, cfg);
  if (cfg.magic != 0xBEEF) {
    cfg.magic = 0xBEEF;
    cfg.moistureThreshold = 400;
    cfg.tempThresholdC    = 28.0;
    cfg.pumpMaxSeconds    = 30;
    cfg.humThreshold      = 70;
    cfg.autoMode          = false;
    cfg.tempOffset        = 0.0;
    cfg.humOffset         = 0.0;
    cfg.moistOffset       = 0;
    cfg.moistDry          = 700;
    cfg.moistWet          = 300;
    setDefaultsIP();
    cfg.lang = HU;
    EEPROM.put(0, cfg);
    EEPROM.commit();
  }
  // Ha 0.0.0.0 lenne, visszaállítjuk
  bool invalidIP = (cfg.ip[0]==0 && cfg.ip[1]==0 && cfg.ip[2]==0 && cfg.ip[3]==0);
  if (invalidIP) { setDefaultsIP(); EEPROM.put(0,cfg); EEPROM.commit(); }
  autoMode = cfg.autoMode;
}
void saveSettings() {
  cfg.autoMode = autoMode;
  EEPROM.put(0, cfg);
  EEPROM.commit();
}

// --- MQTT reconnect ---
void MQTT_connect() {
  int8_t ret;
  if (mqtt.connected()) return;
  Serial.print(F("Csatlakozás az Adafruit IO-hoz... "));
  uint8_t retries = 3;
  while ((ret = mqtt.connect()) != 0) {
    Serial.println(mqtt.connectErrorString(ret));
    Serial.println(F("Újrapróbálkozás 5 mp múlva..."));
    mqtt.disconnect();
    delay(5000);
    if (--retries == 0) {
      ESP.restart();
    }
  }
  Serial.println(F("MQTT kapcsolat létrejött!"));
}

// --- Mentés és restart ---
void handleSave() {
  if (!authenticated) {
    server.send(403, "text/plain; charset=utf-8", t("no_perm"));
    return;
  }
  if (server.hasArg("moist")) cfg.moistureThreshold = constrain(server.arg("moist").toInt(), 0, 1023);
  if (server.hasArg("temp"))  cfg.tempThresholdC    = server.arg("temp").toFloat();
  if (server.hasArg("pump"))  cfg.pumpMaxSeconds    = constrain(server.arg("pump").toInt(), 5, 600);
  if (server.hasArg("hum"))   cfg.humThreshold      = constrain(server.arg("hum").toInt(), 0, 100);
  if (server.hasArg("moistDry")) cfg.moistDry = constrain(server.arg("moistDry").toInt(), 0, 1023);
  if (server.hasArg("moistWet")) cfg.moistWet = constrain(server.arg("moistWet").toInt(), 0, 1023);
  if (cfg.moistWet > cfg.moistDry) {
    int mid = (cfg.moistWet + cfg.moistDry) / 2;
    cfg.moistWet = max(0, mid - 10);
    cfg.moistDry = min(1023, mid + 10);
  }
  if (server.hasArg("tempOffset")) cfg.tempOffset = server.arg("tempOffset").toFloat();
  if (server.hasArg("humOffset"))  cfg.humOffset  = server.arg("humOffset").toFloat();
  if (server.hasArg("moistOffset")) cfg.moistOffset = server.arg("moistOffset").toInt();
  if (server.hasArg("ip")) sscanf(server.arg("ip").c_str(), "%hhu.%hhu.%hhu.%hhu", &cfg.ip[0], &cfg.ip[1], &cfg.ip[2], &cfg.ip[3]);
  if (server.hasArg("gw")) sscanf(server.arg("gw").c_str(), "%hhu.%hhu.%hhu.%hhu", &cfg.gw[0], &cfg.gw[1], &cfg.gw[2], &cfg.gw[3]);
  if (server.hasArg("sn")) sscanf(server.arg("sn").c_str(), "%hhu.%hhu.%hhu.%hhu", &cfg.sn[0], &cfg.sn[1], &cfg.sn[2], &cfg.sn[3]);
  if (server.hasArg("lang")) cfg.lang = (uint8_t) constrain(server.arg("lang").toInt(), 0, 1);

  saveSettings();

  server.send(200, "text/plain; charset=utf-8", "Mentve. Újraindítás...");
  delay(300);
  ESP.restart();
}

// --- JSON adat endpoint ---
void handleData() {
  int moistRaw = analogRead(A0);
  int moistCal = moistRaw + cfg.moistOffset;
  moistCal = constrain(moistCal, 0, 1023);
  int lightRaw = digitalRead(LIGHTPIN);
  int light = (lightRaw == LOW) ? 1 : 0;
  float tCal = (isnan(temp) ? 0 : temp) + cfg.tempOffset;
  float hCal = (isnan(hum) ? 0 : hum) + cfg.humOffset;

  String json = F("{");
  json += "\"temp\":"  + String(tCal,1) + ",";
  json += "\"hum\":"   + String(hCal,0) + ",";
  json += "\"moist\":" + String(moistCal) + ",";
  json += "\"light\":" + String(light);
  json += "}";
  server.send(200, "application/json", json);
}

    if (!pumpState) { pumpState = true; pumpStart = millis(); digitalWrite(PUMP_PIN, RELAY_ON); }
    else { pumpState = false; digitalWrite(PUMP_PIN, RELAY_OFF); }
  }
  server.sendHeader("Location", "/"); server.send(303);
}
void handleFanToggle() {
  if (!autoMode) {
    fanState = !fanState; digitalWrite(FAN_PIN, fanState ? RELAY_ON : RELAY_OFF);
  }
  server.sendHeader("Location", "/"); server.send(303);
}
void handleReset() {
  if (!authenticated) {
    String html = F("<!DOCTYPE html><html><head><meta charset='utf-8'><title>");
    html += t("no_perm");
    html += F("</title><meta name='viewport' content='width=device-width, initial-scale=1'>");
    html += mobileCSS();
    html += F("</head><body><h1>");
    html += t("no_perm");
    html += F("</h1><a class='link' href='/login'>");
    html += t("login_btn_menu");
    html += F("</a></body></html>");
    server.send(403, "text/html; charset=utf-8", html);
    return;
  }
  server.send(200, "text/plain; charset=utf-8", "Újraindítás...");
  delay(200);
  ESP.restart();
}

// --- Soros parancsok: switch–case ---
void handleSerialCommand(const String &cmd) {
  if (cmd.length() == 0) return;
  switch (cmd[0]) {
    case 'P':
      if (cmd == "P1") { pumpState = true; pumpStart = millis(); digitalWrite(PUMP_PIN, RELAY_ON); Serial.println(F("Pumpa BE")); }
      else if (cmd == "P0") { pumpState = false; digitalWrite(PUMP_PIN, RELAY_OFF); Serial.println(F("Pumpa KI")); }
      else if (cmd.startsWith("PM")) { cfg.pumpMaxSeconds = constrain(cmd.substring(2).toInt(), 5, 600); saveSettings(); Serial.printf("PumpMaxSeconds=%d\n", cfg.pumpMaxSeconds); }
      else Serial.println(F("Ismeretlen P parancs"));
      break;
    case 'F':
      if (cmd == "F1") { fanState = true; digitalWrite(FAN_PIN, RELAY_ON); Serial.println(F("Venti BE")); }
      else if (cmd == "F0") { fanState = false; digitalWrite(FAN_PIN, RELAY_OFF); Serial.println(F("Venti KI")); }
      else Serial.println(F("Ismeretlen F parancs"));
      break;
    case 'A':
      if (cmd == "AM") { autoMode = true; saveSettings(); Serial.println(F("Automata mód")); }
      else Serial.println(F("Ismeretlen A parancs"));
      break;
    case 'M':
      if (cmd == "MM") { autoMode = false; saveSettings(); Serial.println(F("Manuál mód")); }
      else if (cmd.startsWith("MT")) { cfg.moistureThreshold = constrain(cmd.substring(2).toInt(),0,1023); saveSettings(); Serial.printf("MoistureThreshold=%d\n", cfg.moistureThreshold); }
      else Serial.println(F("Ismeretlen M parancs"));
      break;
    case 'T':
      if (cmd.startsWith("TD")) { cfg.tempThresholdC = cmd.substring(2).toFloat(); saveSettings(); Serial.printf("TempThreshold=%.1f\n", cfg.tempThresholdC); }
      else Serial.println(F("Ismeretlen T parancs"));
      break;
    case 'H':
      if (cmd.startsWith("HM")) { cfg.humThreshold = constrain(cmd.substring(2).toInt(),0,100); saveSettings(); Serial.printf("HumThreshold=%d\n", cfg.humThreshold); }
      else if (cmd.startsWith("OH")) { cfg.humOffset = cmd.substring(2).toFloat(); saveSettings(); Serial.printf("HumOffset=%.1f\n", cfg.humOffset); }
      else Serial.println(F("Ismeretlen H parancs"));
      break;
    case 'O':
      if (cmd.startsWith("OD")) { cfg.tempOffset = cmd.substring(2).toFloat(); saveSettings(); Serial.printf("TempOffset=%.1f\n", cfg.tempOffset); }
      else if (cmd.startsWith("OM")) { cfg.moistOffset = cmd.substring(2).toInt(); saveSettings(); Serial.printf("MoistOffset=%d\n", cfg.moistOffset); }
      else Serial.println(F("Ismeretlen O parancs"));
      break;
    case 'D':
      if (cmd.startsWith("DR")) { cfg.moistDry = constrain(cmd.substring(2).toInt(),0,1023); saveSettings(); Serial.printf("Dry=%d\n", cfg.moistDry); }
      else Serial.println(F("Ismeretlen D parancs"));
      break;
    case 'W':
      if (cmd.startsWith("WT")) { cfg.moistWet = constrain(cmd.substring(2).toInt(),0,1023); saveSettings(); Serial.printf("Wet=%d\n", cfg.moistWet); }
      else Serial.println(F("Ismeretlen W parancs"));
      break;
    case 'I':
      if (cmd.startsWith("IP")) {
        sscanf(cmd.substring(2).c_str(), "%hhu.%hhu.%hhu.%hhu", &cfg.ip[0], &cfg.ip[1], &cfg.ip[2], &cfg.ip[3]);
        saveSettings();
        Serial.printf("IP: %d.%d.%d.%d\n", cfg.ip[0], cfg.ip[1], cfg.ip[2], cfg.ip[3]);
        Serial.println(F("Újraindítás az új IP-vel..."));
        delay(300);
        ESP.restart();
      } else Serial.println(F("Ismeretlen I parancs"));
      break;
    case 'G':
      if (cmd.startsWith("GW")) {
        sscanf(cmd.substring(2).c_str(), "%hhu.%hhu.%hhu.%hhu", &cfg.gw[0], &cfg.gw[1], &cfg.gw[2], &cfg.gw[3]);
        saveSettings();
        Serial.printf("Gateway: %d.%d.%d.%d\n", cfg.gw[0], cfg.gw[1], cfg.gw[2], cfg.gw[3]);
      } else Serial.println(F("Ismeretlen G parancs"));
      break;
    case 'S':
      if (cmd.startsWith("SN")) {
        sscanf(cmd.substring(2).c_str(), "%hhu.%hhu.%hhu.%hhu", &cfg.sn[0], &cfg.sn[1], &cfg.sn[2], &cfg.sn[3]);
        saveSettings();
        Serial.printf("Subnet: %d.%d.%d.%d\n", cfg.sn[0], cfg.sn[1], cfg.sn[2], cfg.sn[3]);
      } else Serial.println(F("Ismeretlen S parancs"));
      break;
    case 'L':
      if (cmd == "LGHU") { cfg.lang = HU; saveSettings(); Serial.println(F("Nyelv: Magyar")); }
      else if (cmd == "LGEN") { cfg.lang = EN; saveSettings(); Serial.println(F("Language: English")); }
      else Serial.println(F("Ismeretlen L parancs"));
      break;
    default:
      Serial.println(F("Ismeretlen parancs"));
  }
}

// --- Setup ---
void setup() {
  Serial.begin(115200);

  pinMode(PUMP_PIN, OUTPUT);
  pinMode(FAN_PIN, OUTPUT);
  digitalWrite(PUMP_PIN, RELAY_OFF);
  digitalWrite(FAN_PIN, RELAY_OFF);

  pinMode(LIGHTPIN, INPUT);
  dht.begin();

  loadSettings();

  // WiFi config
  IPAddress ip(cfg.ip[0], cfg.ip[1], cfg.ip[2], cfg.ip[3]);
  IPAddress gw(cfg.gw[0], cfg.gw[1], cfg.gw[2], cfg.gw[3]);
  IPAddress sn(cfg.sn[0], cfg.sn[1], cfg.sn[2], cfg.sn[3]);
  IPAddress dns(8,8,8,8);

  WiFi.mode(WIFI_STA);
  WiFi.config(ip, gw, sn, dns);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("\nWiFi kapcsolódva!");
  Serial.print("Elérhető: http://"); Serial.println(WiFi.localIP());

  // Route-ok
  server.on("/", handleRoot);
  server.on("/data", handleData);
  server.on("/settings", handleSettings);
  server.on("/save", HTTP_POST, handleSave);
  server.on("/auto", handleAuto);
  server.on("/manual", handleManual);
  server.on("/pumptoggle", handlePumpToggle);
  server.on("/fantoggle", handleFanToggle);
  server.on("/reset", handleReset);
  server.on("/login", handleLogin);
  server.on("/dologin", HTTP_POST, handleDoLogin);
  server.on("/help", handleHelp);
  server.begin();

  // MQTT első csatlakozás
  MQTT_connect();
}

// --- Loop ---
void loop() {
  server.handleClient();
  unsigned long now = millis();

  // Szenzorok olvasása 30 mp-enként
  if (now - lastSensorCheck >= sensorInterval) {
    lastSensorCheck = now;
    temp = dht.readTemperature();
    hum  = dht.readHumidity();

    // Automata logika
    if (autoMode) {
      int moistRaw = analogRead(A0);
      int moistCal = moistRaw + cfg.moistOffset;
      moistCal = constrain(moistCal, 0, 1023);

      float tCal = (isnan(temp) ? 0 : temp) + cfg.tempOffset;
      float hCal = (isnan(hum) ? 0 : hum) + cfg.humOffset;

      bool needPump = (moistCal > cfg.moistureThreshold);
      if (needPump && !pumpState) {
        pumpState = true;
        pumpStart = now;
        digitalWrite(PUMP_PIN, RELAY_ON);
      }
      bool desiredFan = (tCal > cfg.tempThresholdC) || (hCal > cfg.humThreshold);
      if (desiredFan != fanState) {
        fanState = desiredFan;
        digitalWrite(FAN_PIN, fanState ? RELAY_ON : RELAY_OFF);
      }
    }
  }

  // Pumpa időkorlát
  if (pumpState) {
    bool timeExceeded = (now - pumpStart) >= (unsigned long)cfg.pumpMaxSeconds * 1000;
    if (timeExceeded) {
      pumpState = false;
      digitalWrite(PUMP_PIN, RELAY_OFF);
    }
  }

  // Soros riport percenként
  static unsigned long lastSerialReport = 0;
  if (millis() - lastSerialReport > 60000) {
    lastSerialReport = millis();
    int moistRaw = analogRead(A0);
    int moistCal = moistRaw + cfg.moistOffset;
    moistCal = constrain(moistCal, 0, 1023);
    int lightRaw = digitalRead(LIGHTPIN);
    int light = (lightRaw == LOW) ? 1 : 0;
    float tCal = (isnan(temp) ? 0 : temp) + cfg.tempOffset;
    float hCal = (isnan(hum) ? 0 : hum) + cfg.humOffset;
    Serial.printf("TEMP: %.1f C, HUM: %.0f %%, MOIST: %d, LIGHT: %s, PUMP: %s, FAN: %s, MODE: %s\n",
      tCal, hCal, moistCal, light ? "Világos" : "Sötét",
      pumpState ? "ON" : "OFF", fanState ? "ON" : "OFF", autoMode ? "AUTO" : "MANUAL");
  }

  // Soros parancs olvasás
  static String serialBuffer;
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n' || c == '\r') {
      if (serialBuffer.length() > 0) {
        handleSerialCommand(serialBuffer);
        serialBuffer = "";
      }
    } else {
      serialBuffer += c;
    }
  }

  // WiFi watchdog
  static unsigned long lastWifiCheck = 0;
  static int reconnectTries = 0;
  if (millis() - lastWifiCheck > 30000) {
    lastWifiCheck = millis();
    if (WiFi.status() != WL_CONNECTED) {
      Serial.println("WiFi kapcsolat elveszett, próbálkozás újracsatlakozással...");
      WiFi.disconnect();
      IPAddress ip(cfg.ip[0], cfg.ip[1], cfg.ip[2], cfg.ip[3]);
      IPAddress gw(cfg.gw[0], cfg.gw[1], cfg.gw[2], cfg.gw[3]);
      IPAddress sn(cfg.sn[0], cfg.sn[1], cfg.sn[2], cfg.sn[3]);
      IPAddress dns(8,8,8,8);
      WiFi.config(ip, gw, sn, dns);
      WiFi.begin(ssid, password);
      reconnectTries++;
      if (reconnectTries >= 3) {
        Serial.println("Nem sikerült újracsatlakozni, újraindítás...");
        ESP.restart();
      }
    } else {
      reconnectTries = 0;
    }
  }

  // MQTT keep-alive + adatküldés percenként
  MQTT_connect();
  mqtt.processPackets(10);
  static unsigned long lastMqtt = 0;
  if (millis() - lastMqtt > 60000) {
    lastMqtt = millis();

    int moistRaw = analogRead(A0);
    int moistCal = moistRaw + cfg.moistOffset;
    moistCal = constrain(moistCal, 0, 1023);
    int lightRaw = digitalRead(LIGHTPIN);
    int light = (lightRaw == LOW) ? 1 : 0;
    float tCal = (isnan(temp) ? 0 : temp) + cfg.tempOffset;
    float hCal = (isnan(hum) ? 0 : hum) + cfg.humOffset;

    if (!feedTemp.publish(tCal))      Serial.println(F("Temp publish FAIL"));
    if (!feedHum.publish(hCal))       Serial.println(F("Hum publish FAIL"));
    if (!feedMoist.publish(moistCal)) Serial.println(F("Moist publish FAIL"));
    if (!feedLight.publish(light))    Serial.println(F("Light publish FAIL"));
    Serial.println(F("Adatok elküldve az Adafruit IO-ra."));
  }

  yield();
}
